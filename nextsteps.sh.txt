cd ~/evidence && python - <<'EOF'
import re, glob, os
files = {p: open(p, errors='ignore').read().replace(',', '')
         for p in glob.glob('system*/**/*', recursive=True) if os.path.isfile(p)}
alltext = '\n'.join(files.values())
for i, line in enumerate(open('reflection-brief-template.md').read().splitlines(), 1):
    cited = re.findall(r'system\d+/[\w./-]*\w', line)
    for c in cited:
        if not os.path.exists(c):
            print(f'line {i}: path {c} does not exist')
    scope = '\n'.join(v for k, v in files.items()
                      if any(k == c or k.startswith(c + '/') for c in cited)) or alltext
    bare = line
    for c in cited:
        bare = bare.replace(c, ' ')
    for n in re.findall(r'\d[\d,]*(?:\.\d+)?', bare):
        n2 = n.replace(',', '')
        if len(n2.replace('.', '')) < 2:
            continue
        if not re.search(r'(?<![\d.])' + re.escape(n2) + r'(?![\d])', scope):
            where = 'its cited file(s)' if cited else 'any evidence file'
            print(f'line {i}: {n} not in {where} | {line.strip()[:160]}')
EOF
